# codigo_intermedio.py — plantilla de variables
DIRECTIVA = {
    "int": "DW",
    "bool": "DB",
    "string": "DB",
    "float": "DD",
}

def _armar(self):
    variables = []
    for nombre, tipo in self.variables:
        directiva = DIRECTIVA.get(tipo)
        if directiva:
            variables.append(
                self._dato(nombre, directiva, "?")
            )
    # →  x       DW      ?
